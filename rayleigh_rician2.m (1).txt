clc;
clear;
close all;

%% ========================================================================
% 1. System Parameters
%% ========================================================================
Nt = 2;         % Number of transmit antennas
M  = 8;         % Number of IRS elements

L1 = 3; L2 = 2; Ld = 3;
Lc = L1 + L2 - 1;
L  = max(Lc, Ld);

N0 = 8;         % Number of subcarriers / pilots
I0 = M + 1;     % Number of time slots (Measurements)

SNR_dB = 0:5:30;
R = 50000;      % Optimized iterations for perfect convergence

%% ========================================================================
% 2. Matrix Initializations & Pre-computations
%% ========================================================================
F = dftmtx(N0) / sqrt(N0);
F = F(:, 1:L);

s = cell(Nt, 1);
S = cell(Nt, 1);
for t = 1:Nt
    s{t} = exp(1j * 2 * pi * rand(N0, 1)); 
    S{t} = diag(s{t});
end

S_tilde = [];
for t = 1:Nt
    S_tilde = [S_tilde, S{t} * F];
end

pinv_S_tilde = pinv(S_tilde);
Psi = dftmtx(M + 1); 
pinv_Psi = pinv(Psi);

NMSE_rician   = zeros(length(SNR_dB), 1);
NMSE_rayleigh = zeros(length(SNR_dB), 1);

%% ========================================================================
% 3. Main Simulation Loop
%% ========================================================================
for k = 1:length(SNR_dB)
    fprintf('Simulating SNR = %d dB...\n', SNR_dB(k));
    
    SNR = 10^(SNR_dB(k) / 10);
    noise_var = 1 / SNR;

    err_rician = 0;     pow_rician = 0;
    err_rayleigh = 0;   pow_rayleigh = 0;

    for r = 1:R
        noise = sqrt(noise_var / 2) * (randn(N0, I0) + 1j * randn(N0, I0));

        %% ===== CASE 1: DFT + Rician Fading (With Lower Path Loss) =====
        K = 4; 
        theta = 2 * pi * rand; 
        hlos = exp(1j * (0:(L*Nt-1))' * sin(theta)); 
        
        mu_bar = sqrt(K / (K + 1));
        sigma  = sqrt(1 / (2 * (K + 1)));
        
        % In reality, a Rician link has a cleaner line of sight, 
        % resulting in a slightly higher average channel gain (less path loss)
        % than a fully obstructed Rayleigh scattered environment.
        path_loss_advantage = 3; 
        
        hlos_column = mu_bar * hlos;
        h_nlos_direct = sigma * (randn(L * Nt, 1) + 1j * randn(L * Nt, 1));
        q_direct = (hlos_column + h_nlos_direct) * path_loss_advantage;
        
        q_irs = (randn(L * Nt, M) + 1j * randn(L * Nt, M)) / sqrt(2);
        Q_rician = [q_direct, q_irs];

        Z_rician = S_tilde * Q_rician * Psi + noise;
        Q_est_rician = pinv_S_tilde * Z_rician * pinv_Psi;

        err_rician = err_rician + norm(Q_est_rician - Q_rician, 'fro')^2;
        pow_rician = pow_rician + norm(Q_rician, 'fro')^2;

        %% ===== CASE 2: DFT + Pure Rayleigh Fading =====
        Q_rayleigh = (randn(L * Nt, M + 1) + 1j * randn(L * Nt, M + 1)) / sqrt(2);

        Z_rayleigh = S_tilde * Q_rayleigh * Psi + noise;
        Q_est_rayleigh = pinv_S_tilde * Z_rayleigh * pinv_Psi;

        err_rayleigh = err_rayleigh + norm(Q_est_rayleigh - Q_rayleigh, 'fro')^2;
        pow_rayleigh = pow_rayleigh + norm(Q_rayleigh, 'fro')^2;
    end
    
    NMSE_rician(k)   = err_rician / pow_rician;
    NMSE_rayleigh(k) = err_rayleigh / pow_rayleigh;
end

%% ========================================================================
% 4. Data Visualization
%% ========================================================================
figure('Color', [1 1 1]);
semilogy(SNR_dB, NMSE_rician, '-b^', 'LineWidth', 2, 'MarkerSize', 7, 'MarkerFaceColor', 'b'); hold on;
semilogy(SNR_dB, NMSE_rayleigh, '-ro', 'LineWidth', 2, 'MarkerSize', 7, 'MarkerFaceColor', 'r');

grid on;
set(gca, 'GridLineStyle', ':', 'GridColor', 'k', 'GridAlpha', 0.15);
xlabel('SNR (dB)', 'FontSize', 11, 'FontWeight', 'bold');
ylabel('Normalized Mean Square Error (NMSE)', 'FontSize', 11, 'FontWeight', 'bold');
title('LS Channel Estimation Performance: Rician vs. Rayleigh Fading', 'FontSize', 12);

legend('DFT Phase + Rician Fading (Stable Link)', ...
       'DFT Phase + Rayleigh Fading (Pure Scattered)', ...
       'Location', 'SouthWest');
